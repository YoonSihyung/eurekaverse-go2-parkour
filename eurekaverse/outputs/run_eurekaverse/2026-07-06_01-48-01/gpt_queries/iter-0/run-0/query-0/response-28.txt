import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A repeated balance-beam course with narrow raised planks over pits, testing precision foot placement and straight-line traversal."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # ---------------------------------------------------------------------
    # Course idea:
    # The robot must cross 6 narrow raised planks separated by gaps/pits.
    # Each plank is centered along the corridor and slightly elevated.
    # This tests:
    #   - precise foot placement on narrow supports
    #   - stable straight-line walking
    #   - stepping across gaps without side deviation
    #
    # The course stays consistent: repeated beam-like obstacles, no turns.
    # The spawn area is kept flat and clear for the first 2 meters.
    # ---------------------------------------------------------------------

    L = m_to_idx(length)   # total x extent, expected 240
    W = m_to_idx(width)    # total y extent, expected 80
    mid_y = W // 2

    # Keep a safe flat spawn zone; obstacles begin after 2 m
    spawn_end = m_to_idx(2.0)
    height_field[:spawn_end, :] = 0.0

    # Difficulty-scaled parameters
    # Narrow beam width is challenging but still realistic for a quadruped.
    beam_width_m = 0.55 - 0.10 * difficulty          # [0.45, 0.55] m
    beam_width = max(m_to_idx(beam_width_m), m_to_idx(0.40))

    beam_length_m = 1.10 - 0.15 * difficulty         # [0.95, 1.10] m
    beam_length = max(m_to_idx(beam_length_m), m_to_idx(0.40))

    beam_height = 0.08 + 0.10 * difficulty           # slightly higher at harder settings
    gap_length_m = 0.45 + 0.55 * difficulty          # [0.45, 1.0] m
    gap_length = m_to_idx(gap_length_m)

    # Small lateral offsets to require minor correction but keep overall straight path.
    # These offsets are always within the course bounds and centered around the corridor.
    max_side_shift_m = 0.15 + 0.10 * difficulty
    max_side_shift = m_to_idx(max_side_shift_m)

    # Helper to clamp indices into valid terrain bounds
    def clamp(v, lo, hi):
        return max(lo, min(hi, v))

    # Build repeated planks with pits in between
    cur_x = spawn_end
    plank_centers = []

    for i in range(6):
        # Random but bounded lateral variation
        dy = random.randint(-max_side_shift, max_side_shift)

        # Slight length variation for natural-looking structure
        local_beam_length = beam_length + random.randint(-m_to_idx(0.10), m_to_idx(0.10))
        local_beam_length = max(local_beam_length, m_to_idx(0.40))

        x1 = cur_x
        x2 = min(cur_x + local_beam_length, L)

        # Center beam around the middle of the lane
        y_center = clamp(mid_y + dy, beam_width // 2 + 1, W - beam_width // 2 - 2)
        y1 = clamp(y_center - beam_width // 2, 0, W)
        y2 = clamp(y_center + beam_width // 2, 0, W)

        # Create the raised plank
        height_field[x1:x2, y1:y2] = beam_height

        # Put the goal near the center of each plank
        goal_x = x1 + (x2 - x1) // 2
        goal_y = y_center
        goals[i] = [goal_x, goal_y]

        plank_centers.append((goal_x, goal_y))

        # Create the pit after each plank, except after the last one
        cur_x = x2 + gap_length

        # Fill pit with negative height to prevent easy stepping down/up transitions
        pit_x1 = x2
        pit_x2 = min(x2 + gap_length, L)
        if i < 5:
            height_field[pit_x1:pit_x2, :] = -0.7 - 0.3 * difficulty

    # Ensure the remaining area after the last plank is flat so the robot can finish
    if cur_x < L:
        height_field[cur_x:, :] = 0.0

    # Final goal placed on flat ground beyond the last gap
    final_goal_x = min(L - m_to_idx(0.5), cur_x + m_to_idx(0.6))
    goals[6] = [final_goal_x, mid_y]
    goals[7] = [min(L - 1, final_goal_x + m_to_idx(0.4)), mid_y]

    # Make sure goals are within bounds and integer-valued indices
    goals[:, 0] = np.clip(np.round(goals[:, 0]), 0, L - 1)
    goals[:, 1] = np.clip(np.round(goals[:, 1]), 0, W - 1)

    return height_field, goals
