```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A straight-line course of alternating raised beams and shallow gaps for precise balance and gap crossing."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Initialize terrain
    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # Map dimensions in indices
    L = m_to_idx(length)
    W = m_to_idx(width)
    mid_y = W // 2

    # Keep a safe flat spawn zone so the robot starts cleanly
    spawn_end = m_to_idx(2.0)
    height_field[:spawn_end, :] = 0.0

    # Choose a consistent obstacle style: narrow elevated beams separated by pits
    # This tests balance + controlled stepping/jumping along a straight corridor.
    # Difficulty increases beam height, beam length variability, and pit depth/length.
    beam_h_min = 0.02 + 0.05 * difficulty
    beam_h_max = 0.06 + 0.12 * difficulty
    pit_depth = -(0.25 + 0.55 * difficulty)

    beam_len_base = 0.85 - 0.18 * difficulty
    beam_len_jitter = 0.18 - 0.08 * difficulty
    gap_len_base = 0.35 + 0.55 * difficulty
    gap_len_jitter = 0.15 + 0.10 * difficulty

    # Beam width: keep at least 1m, but not too wide so the robot must stay centered
    beam_half_w = max(m_to_idx(0.55), m_to_idx(0.50 + 0.05 * difficulty))

    # Place 7 obstacle segments after the spawn; goals go on/near each beam and the final landing
    cur_x = spawn_end + m_to_idx(0.3)
    goal_i = 0

    # First goal slightly before first beam, still on flat ground
    goals[goal_i] = [spawn_end - m_to_idx(0.3), mid_y]
    goal_i += 1

    for seg in range(6):
        # Beam dimensions with small controlled randomness
        beam_len = beam_len_base + random.uniform(-beam_len_jitter, beam_len_jitter)
        beam_len = max(0.55, beam_len)
        beam_len_idx = m_to_idx(beam_len)

        # Keep beams within bounds and leave space for the last landing zone
        if cur_x + beam_len_idx >= L - m_to_idx(1.2):
            beam_len_idx = max(m_to_idx(0.6), L - m_to_idx(1.2) - cur_x)

        # Gentle y-offsets to force minor lateral correction while staying straight-line overall
        y_offset_m = random.uniform(-0.15, 0.15) * (0.4 + 0.6 * difficulty)
        y_center = int(np.clip(mid_y + m_to_idx(y_offset_m), beam_half_w + 1, W - beam_half_w - 2))

        # Raise the beam
        beam_height = random.uniform(beam_h_min, beam_h_max)
        x1, x2 = cur_x, cur_x + beam_len_idx
        y1, y2 = y_center - beam_half_w, y_center + beam_half_w
        height_field[x1:x2, y1:y2] = beam_height

        # Place goal near the center of the beam
        goals[goal_i] = [x1 + beam_len_idx / 2.0, y_center]
        goal_i += 1

        # Create a pit after the beam to prevent simply dropping down and re-ascending
        gap_len = gap_len_base + random.uniform(-gap_len_jitter, gap_len_jitter)
        gap_len = max(0.25, gap_len)
        gap_len_idx = m_to_idx(gap_len)

        pit_x1 = x2
        pit_x2 = min(L, x2 + gap_len_idx)
        height_field[pit_x1:pit_x2, :] = pit_depth

        # Make a narrow safe approach strip at the beam edges to avoid harsh vertical discontinuities
        if x1 > 0:
            height_field[max(0, x1 - m_to_idx(0.1)):x1, y1:y2] = 0.0
        if x2 < L:
            height_field[x2:min(L, x2 + m_to_idx(0.1)), y1:y2] = 0.0

        cur_x = pit_x2

    # Final landing zone: flat ground with the last goal centered on it
    if cur_x < L:
        height_field[cur_x:, :] = 0.0

    # Make sure the last goal is on the final flat section
    goals[-1] = [min(L - 2, cur_x + m_to_idx(0.6)), mid_y]

    # Fill any unassigned intermediate goals safely if needed
    for k in range(goal_i, 7):
        frac = (k - goal_i + 1) / max(1, 7 - goal_i + 1)
        goals[k] = [spawn_end + frac * (L - spawn_end - m_to_idx(1.0)), mid_y]

    # Clip goals to valid indices
    goals[:, 0] = np.clip(goals[:, 0], 0, L - 1)
    goals[:, 1] = np.clip(goals[:, 1], 0, W - 1)

    return height_field, goals
```
