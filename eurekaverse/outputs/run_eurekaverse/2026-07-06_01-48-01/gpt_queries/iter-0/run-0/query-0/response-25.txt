```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A straight sequence of raised balance beams over pits, testing precise foot placement and gap crossing."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Terrain dimensions in quantized cells
    H = m_to_idx(length)
    W = m_to_idx(width)
    height_field = np.zeros((H, W), dtype=np.float32)
    goals = np.zeros((8, 2), dtype=np.float32)

    # Course layout:
    # - Flat spawn zone
    # - A sequence of narrow raised beams separated by pits
    # - The robot must keep a straight heading while stepping/jumping between beams
    # This emphasizes balance and gap crossing without requiring turns.

    # Fixed course centerline
    mid_y = W // 2

    # Spawn zone: keep first 2 m flat so the robot starts unobstructed
    spawn_len = m_to_idx(2.0)
    height_field[:spawn_len, :] = 0.0

    # Difficulty-scaled parameters
    beam_length_m = 0.85 - 0.15 * difficulty          # slightly shorter at harder settings
    beam_length = max(m_to_idx(0.45), m_to_idx(beam_length_m))
    beam_width_m = 0.55 + 0.15 * difficulty           # still narrow, but realistic for a quadruped
    beam_half_width = max(m_to_idx(0.20), m_to_idx(beam_width_m) // 2)

    beam_height = 0.08 + 0.18 * difficulty            # raised enough to matter, not too tall
    pit_depth = -0.35 - 0.55 * difficulty             # deeper pits at higher difficulty

    # Gap between beams grows with difficulty
    gap_m = 0.35 + 0.45 * difficulty
    gap_len = max(m_to_idx(0.25), m_to_idx(gap_m))

    # Small lateral drift to force careful alignment while still going straight overall
    y_drift_m = 0.25 * difficulty
    y_choices = [
        0.0,
        0.6 * y_drift_m,
        -0.6 * y_drift_m,
        1.0 * y_drift_m,
        -1.0 * y_drift_m,
        0.6 * y_drift_m,
        -0.6 * y_drift_m,
        0.0,
    ]

    # Clear everything after spawn to a pit first; beams will be painted on top
    height_field[spawn_len:, :] = pit_depth

    # Helper to paint one beam
    def add_beam(x1, x2, y_center_idx):
        y1 = max(0, y_center_idx - beam_half_width)
        y2 = min(W, y_center_idx + beam_half_width + 1)
        x1 = max(0, x1)
        x2 = min(H, x2)
        height_field[x1:x2, y1:y2] = beam_height

    # Place goals:
    # goal 0 near the end of spawn
    goals[0] = [spawn_len - m_to_idx(0.4), mid_y]

    cur_x = spawn_len + m_to_idx(0.25)

    # Create 7 beam segments, with goals on each one and a final goal after the last beam
    for i in range(7):
        y_offset = m_to_idx(y_choices[i])
        beam_center_y = int(np.clip(mid_y + y_offset, beam_half_width + 1, W - beam_half_width - 2))

        x1 = cur_x
        x2 = cur_x + beam_length
        add_beam(x1, x2, beam_center_y)

        # Place the goal near the center of the beam
        goals[i + 1] = [x1 + beam_length * 0.5, beam_center_y]

        cur_x = x2 + gap_len

    # Final goal after the last gap, on flat ground
    # Keep the end region flat so the robot can step off the last beam cleanly
    end_x = min(H, cur_x + m_to_idx(0.8))
    if cur_x < H:
        height_field[cur_x:end_x, :] = 0.0

    goals[7] = [min(H - 1, cur_x + m_to_idx(0.4)), int(np.clip(mid_y, 0, W - 1))]

    return height_field, goals
```
