```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A sequence of raised balance beams and short gaps that trains careful straight-line stepping and precise foot placement."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # Terrain dimensions in indices
    L = m_to_idx(length)
    W = m_to_idx(width)
    mid_y = W // 2

    # Keep the first 2 meters flat for spawn safety
    spawn_end = m_to_idx(2.0)
    height_field[:spawn_end, :] = 0.0

    # Difficulty-scaled beam parameters
    beam_height = 0.08 + 0.12 * difficulty          # 8 cm to 20 cm
    beam_width_m = 0.55 + 0.15 * (1.0 - difficulty) # wider at low difficulty
    beam_width = max(m_to_idx(beam_width_m), m_to_idx(0.4))  # rare narrow allowance honored
    beam_length_m = 0.85 + 0.15 * (1.0 - difficulty) # long enough to require commitment
    beam_length = m_to_idx(beam_length_m)

    # Gap parameters
    gap_m = 0.18 + 0.55 * difficulty
    gap = m_to_idx(gap_m)

    # Small lateral offsets to require micro-corrections while staying mostly straight
    max_offset = m_to_idx(0.18 + 0.12 * difficulty)
    offset_choices = [-max_offset, -max_offset // 2, 0, max_offset // 2, max_offset]

    def add_beam(x_start, x_end, y_center):
        """Place a rectangular raised beam."""
        x1 = max(0, min(L, x_start))
        x2 = max(0, min(L, x_end))
        half_w = beam_width // 2
        y1 = max(0, y_center - half_w)
        y2 = min(W, y_center + half_w + 1)
        height_field[x1:x2, y1:y2] = beam_height

    def add_pit(x_start, x_end):
        """Carve a pit below ground to force the robot onto the beam."""
        x1 = max(0, min(L, x_start))
        x2 = max(0, min(L, x_end))
        height_field[x1:x2, :] = -0.9

    # Build 8 goal locations and 7 beam segments between/around them
    cur_x = spawn_end + m_to_idx(0.4)

    # First goal on the first beam
    first_offset = random.choice(offset_choices)
    first_y = int(np.clip(mid_y + first_offset, beam_width // 2, W - beam_width // 2 - 1))
    add_beam(cur_x, cur_x + beam_length, first_y)
    goals[0] = [cur_x + beam_length // 2, first_y]

    cur_x += beam_length

    for i in range(1, 8):
        # Gap before next beam
        if i < 8:
            add_pit(cur_x, cur_x + gap)
            cur_x += gap

        if i < 8:
            # Alternate slight offsets to create subtle lateral stepping demands
            offset = offset_choices[i % len(offset_choices)]
            # Make the path gently vary but remain traversable in a straight-line instruction sense
            y_center = int(np.clip(mid_y + offset, beam_width // 2, W - beam_width // 2 - 1))

            # Ensure the beam fits in the remaining terrain
            end_x = min(cur_x + beam_length, L - 1)
            add_beam(cur_x, end_x, y_center)

            # Place goal near the center of the beam
            goal_x = cur_x + (end_x - cur_x) // 2
            goals[i] = [goal_x, y_center]

            cur_x = end_x

    # Make sure the area after the last beam is flat so the robot can finish cleanly
    if cur_x < L:
        height_field[cur_x:, :] = 0.0

    # Final sanity clamp for all goals to stay in bounds
    goals[:, 0] = np.clip(goals[:, 0], 0, L - 1)
    goals[:, 1] = np.clip(goals[:, 1], 0, W - 1)

    return height_field, goals
```
