import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A slalom of low balance beams and pits that tests precise foot placement and lateral stability."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Terrain grid
    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2), dtype=np.int16)

    # -------------------------------------------------------------------------
    # Course idea:
    # - The robot starts on flat ground.
    # - It then crosses a series of narrow raised beams alternating left/right.
    # - Between beams are shallow pits, forcing the robot to stay on top rather
    #   than step down and back up.
    # - Goals are placed on the center of each beam, with the beam direction
    #   staying mostly straight down the course, but with lateral offsets that
    #   require controlled side-to-side traversal.
    # - This tests balance, precise foot placement, and recovery from small
    #   lateral shifts.
    # -------------------------------------------------------------------------

    n_x = m_to_idx(length)
    n_y = m_to_idx(width)
    mid_y = n_y // 2

    # Spawn-safe flat region
    spawn_len = m_to_idx(2.0)
    height_field[:spawn_len, :] = 0.0

    # Keep the rest of the course as a pit so the robot must use the beams
    height_field[spawn_len:, :] = -0.9 - 0.2 * difficulty

    # Beam and gap parameters
    beam_len = m_to_idx(1.15 - 0.15 * difficulty)   # shorter and harder at higher difficulty
    gap_len = m_to_idx(0.55 + 0.35 * difficulty)    # larger gaps as difficulty increases
    beam_w_min = 0.45
    beam_w_max = 0.75 - 0.1 * difficulty
    beam_w = m_to_idx(np.clip(np.random.uniform(beam_w_min, beam_w_max), 0.4, None))

    # Beam height rises slightly with difficulty, but remains realistic for a quadruped
    beam_h = 0.05 + 0.18 * difficulty

    # Lateral offsets: alternate left/right around center
    lateral_step_min = 0.45
    lateral_step_max = 1.0 - 0.15 * difficulty
    max_offset = m_to_idx(min(lateral_step_max, (width / 2) - 0.55))
    min_offset = m_to_idx(lateral_step_min)
    lateral_choices = [min_offset, -min_offset, max_offset, -max_offset]

    def add_beam(x1, x2, center_y, beam_width, height):
        """Adds a raised rectangular beam to the terrain."""
        half_w = beam_width // 2
        y1 = max(0, center_y - half_w)
        y2 = min(n_y, center_y + half_w + 1)
        x1 = max(0, x1)
        x2 = min(n_x, x2)
        height_field[x1:x2, y1:y2] = height

    # Place first goal near the end of the spawn area
    goals[0] = [spawn_len - m_to_idx(0.35), mid_y]

    cur_x = spawn_len
    cur_y = mid_y

    # We will place 6 beams and use the final goal after the last beam
    for i in range(6):
        # Alternate lateral target position
        if i % 2 == 0:
            cur_y = mid_y + lateral_choices[i % len(lateral_choices)]
        else:
            cur_y = mid_y - lateral_choices[i % len(lateral_choices)]

        # Keep beam fully inside bounds
        cur_y = int(np.clip(cur_y, beam_w // 2 + 1, n_y - beam_w // 2 - 2))

        # Add the beam
        add_beam(cur_x, cur_x + beam_len, cur_y, beam_w, beam_h)

        # Goal centered on the beam
        goal_x = cur_x + beam_len // 2
        goals[i + 1] = [goal_x, cur_y]

        # Advance by beam + gap
        cur_x += beam_len + gap_len

    # Final goal at the end of the last gap, back on flat ground
    final_x = min(n_x - 2, cur_x + m_to_idx(0.45))
    goals[7] = [final_x, mid_y]

    # Ensure area after the last beam is traversable flat ground
    height_field[cur_x:, :] = -0.9 - 0.2 * difficulty
    height_field[final_x:, :] = 0.0

    # Slightly smooth the landing region for realism
    landing_len = m_to_idx(0.6)
    landing_start = max(0, final_x - landing_len)
    height_field[landing_start:final_x + 1, :] = 0.0

    return height_field, goals
