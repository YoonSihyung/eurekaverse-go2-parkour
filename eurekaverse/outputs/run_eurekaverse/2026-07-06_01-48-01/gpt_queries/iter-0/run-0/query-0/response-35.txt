```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """Stepping-stone river with alternating narrow planks to test precise foot placement and balance."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Terrain grid
    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    n_x, n_y = height_field.shape
    mid_y = n_y // 2

    # Spawn-safe flat region
    spawn_len = m_to_idx(2.0)
    height_field[:spawn_len, :] = 0.0

    # Everything beyond spawn is a "river" pit, forcing the robot to use the stepping stones
    pit_depth = -0.65 - 0.35 * difficulty
    height_field[spawn_len:, :] = pit_depth

    # Skill focus: precise, repeated stepping over narrow elevated planks with slight lateral offsets.
    # Obstacles are realistic: like park agility planks or urban crossing beams.
    plank_len = m_to_idx(0.85 - 0.15 * difficulty)   # long enough to stand on, but not too easy
    plank_wid = m_to_idx(1.10 - 0.15 * difficulty)   # >= 1m wide as requested
    plank_h = 0.10 + 0.10 * difficulty                # elevated above the pit
    gap_len = m_to_idx(0.75 + 0.35 * difficulty)

    # Lateral offset pattern to force subtle side-to-side adjustments
    y_offsets_m = [0.0, 0.25, -0.25, 0.35, -0.35, 0.20, -0.20, 0.0]
    y_offsets = [m_to_idx(v) for v in y_offsets_m]

    # Place eight goal locations, one on each traversable segment.
    # The quadruped is instructed to move straight between successive goals, so
    # we keep the course mostly forward with gentle lateral shifts.
    cur_x = spawn_len + m_to_idx(0.25)

    for i in range(8):
        # Clamp obstacle dimensions so they fit within the terrain bounds
        x1 = int(max(cur_x, 0))
        x2 = int(min(cur_x + plank_len, n_x))
        cy = int(np.clip(mid_y + y_offsets[i], plank_wid // 2, n_y - plank_wid // 2 - 1))
        y1 = int(max(cy - plank_wid // 2, 0))
        y2 = int(min(cy + plank_wid // 2, n_y))

        # Create the plank by raising it above the pit
        height_field[x1:x2, y1:y2] = plank_h

        # Slightly taper the edges for realism and smoother contacts
        edge_pad = max(1, m_to_idx(0.08))
        if x2 - x1 > 2 * edge_pad and y2 - y1 > 2 * edge_pad:
            height_field[x1:x1 + edge_pad, y1:y2] = plank_h * 0.6
            height_field[x2 - edge_pad:x2, y1:y2] = plank_h * 0.6
            height_field[x1:x2, y1:y1 + edge_pad] = plank_h * 0.6
            height_field[x1:x2, y2 - edge_pad:y2] = plank_h * 0.6

        # Place goal near the center of each plank
        goals[i] = [x1 + (x2 - x1) / 2, cy]

        # Advance to next plank with a gap
        cur_x = x2 + gap_len

    # Ensure the terrain after the last plank stays pit-like, but the very end can be flat for termination
    end_flat = m_to_idx(0.6)
    if n_x - cur_x > end_flat:
        height_field[-end_flat:, :] = 0.0

    # Keep the spawn area and initial approach flat and safe
    height_field[:spawn_len, :] = 0.0

    return height_field, goals
```
