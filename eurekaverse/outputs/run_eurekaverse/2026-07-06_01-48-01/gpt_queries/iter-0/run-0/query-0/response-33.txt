```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A repeating low-wall slalom course with narrow lateral offsets and short jumps."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Create terrain arrays
    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # Terrain dimensions in indices
    L = m_to_idx(length)
    W = m_to_idx(width)

    # Useful constants
    mid_y = W // 2
    spawn_x = m_to_idx(2.0)  # keep the first 2 m obstacle-free
    wall_h = 0.18 + 0.22 * difficulty  # moderate step height, increases with difficulty
    pit_h = -0.55 - 0.35 * difficulty  # negative terrain to discourage stepping off the intended path

    # Course structure:
    # - The robot runs forward through 7 repeated "slalom gate" segments.
    # - Each segment has a raised low wall spanning most of the width, with an alternating gap
    #   on the left/right side so the robot must shift laterally while maintaining forward motion.
    # - The area outside the intended path is lowered to a pit to prevent easy bypassing.
    # - This tests balance, lateral foot placement, and turning precision.

    # Make the spawn area flat and safe
    height_field[:spawn_x, :] = 0.0

    # Use a narrow safe corridor near the center that alternates side openings
    segment_len = m_to_idx(1.35 - 0.15 * difficulty)
    wall_thickness = m_to_idx(0.35)
    gap_width = m_to_idx(1.15 + 0.10 * (1.0 - difficulty))  # at least 1 m wide
    corridor_half = m_to_idx(0.55)

    # Helper to clamp indices
    def clamp(v, lo, hi):
        return max(lo, min(hi, v))

    # Start building the slalom segments after the spawn zone
    cur_x = spawn_x + m_to_idx(0.4)

    # First goal: just before the first wall
    goals[0] = [clamp(cur_x - m_to_idx(0.2), 0, L - 1), mid_y]

    for i in range(7):
        # Alternate the opening from left to right to force side-to-side motion
        open_left = (i % 2 == 0)

        # Slightly vary the gate position with difficulty while keeping it realistic
        lateral_shift = m_to_idx((0.25 + 0.15 * difficulty) * (1 if open_left else -1))
        gate_center = clamp(mid_y + lateral_shift, gap_width // 2 + 1, W - gap_width // 2 - 2)

        seg_start = cur_x
        seg_end = clamp(cur_x + segment_len, 0, L)

        # Fill the whole segment with pit, then add a raised wall and a traversable opening
        height_field[seg_start:seg_end, :] = pit_h

        # Build the low wall across most of the width
        wall_y1 = 0
        wall_y2 = W

        # Leave a lateral gap on one side
        if open_left:
            gap_y1 = 1
            gap_y2 = clamp(gate_center + gap_width // 2, 0, W)
            # wall occupies the rest of the width except the opening
            if gap_y2 < W:
                height_field[seg_start:seg_end, gap_y2:W] = wall_h
            # keep a flat approach corridor into the gap
            height_field[seg_start:seg_end, gap_y1:gap_y2] = 0.0
        else:
            gap_y1 = clamp(gate_center - gap_width // 2, 0, W)
            gap_y2 = W - 1
            if gap_y1 > 0:
                height_field[seg_start:seg_end, 0:gap_y1] = wall_h
            height_field[seg_start:seg_end, gap_y1:gap_y2 + 1] = 0.0

        # Add a slightly raised lip at the centerline of the gate to make the crossing precise
        lip_w = m_to_idx(0.18 + 0.05 * difficulty)
        lip_y1 = clamp(gate_center - lip_w // 2, 0, W)
        lip_y2 = clamp(gate_center + lip_w // 2 + 1, 0, W)
        height_field[seg_start:seg_end, lip_y1:lip_y2] = max(0.02, wall_h * 0.15)

        # Goal positioned near the middle of the opening, just beyond the segment
        goal_x = clamp(seg_start + segment_len // 2, 0, L - 1)
        goal_y = gate_center
        goals[i + 1] = [goal_x, goal_y]

        # Advance to the next segment with a short flat landing zone
        landing = m_to_idx(0.25)
        cur_x = seg_end + landing

        # Insert a short flat strip between gates so the robot must stabilize before the next turn
        flat_end = clamp(cur_x + m_to_idx(0.18), 0, L)
        height_field[seg_end:flat_end, :] = 0.0
        cur_x = flat_end

    # Final goal near the end of the course on flat ground
    final_goal_x = clamp(L - m_to_idx(0.6), 0, L - 1)
    goals[7] = [final_goal_x, mid_y]

    # Ensure the final stretch is flat so the robot can complete the course cleanly
    height_field[cur_x:, :] = 0.0

    # Keep the entire outer edge slightly lowered to discourage boundary hugging
    edge_margin = m_to_idx(0.15)
    if edge_margin > 0:
        height_field[:, :edge_margin] = pit_h
        height_field[:, -edge_margin:] = pit_h

    # Re-assert spawn area is flat and safe
    height_field[:spawn_x, :] = 0.0

    return height_field, goals
```
