```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A zigzag sequence of raised balance beams over shallow pits for precise turning and foot placement."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # Terrain dimensions in indices
    L = m_to_idx(length)
    W = m_to_idx(width)

    # Spawn-safe region
    spawn_x = m_to_idx(2.0)
    mid_y = W // 2

    # Keep the start area flat and safe
    height_field[:spawn_x, :] = 0.0

    # Skill focus:
    # Repeated narrow beam traversal with alternating lateral offsets.
    # This forces the robot to maintain balance while making small turns
    # and stepping across short negative pits between beam segments.

    # Beam parameters tuned to stay realistic for a quadruped
    beam_width_m = 0.55 + 0.10 * (1.0 - difficulty)   # >= 0.45 m, slightly easier at low difficulty
    beam_height_m = 0.10 + 0.18 * difficulty          # raised but not overly tall
    pit_depth_m = -0.20 - 0.35 * difficulty           # shallow-to-moderate drop

    beam_width = max(m_to_idx(beam_width_m), m_to_idx(0.4))
    beam_height = beam_height_m
    pit_depth = pit_depth_m

    # Beam segment and gap lengths
    seg_len_m = 1.10 - 0.25 * difficulty
    gap_len_m = 0.35 + 0.55 * difficulty
    seg_len = max(m_to_idx(seg_len_m), m_to_idx(0.4))
    gap_len = max(m_to_idx(gap_len_m), m_to_idx(0.25))

    # Lateral pattern in meters: left, center, right, center, ...
    lateral_offsets_m = [-0.85, -0.25, 0.55, -0.45, 0.75, -0.15, 0.35, 0.0]
    # Slightly vary the offsets with difficulty to make harder courses more precise
    lateral_offsets_m = [o * (0.7 + 0.3 * difficulty) for o in lateral_offsets_m]

    # Helper to place a raised beam segment with surrounding pit
    def add_beam_segment(x1, x2, center_y, seg_width, height):
        half_w = seg_width // 2
        y1 = max(0, center_y - half_w)
        y2 = min(W, center_y + half_w + 1)
        height_field[x1:x2, y1:y2] = height

    # Set all terrain after the spawn zone to a shallow pit by default
    height_field[spawn_x:, :] = pit_depth

    # Build 8 segments total, with goals placed on each segment
    cur_x = spawn_x + m_to_idx(0.25)
    for i in range(8):
        offset_idx = m_to_idx(lateral_offsets_m[i])
        center_y = int(np.clip(mid_y + offset_idx, beam_width // 2, W - beam_width // 2 - 1))

        # Add the beam segment
        x1 = cur_x
        x2 = min(L, cur_x + seg_len)
        add_beam_segment(x1, x2, center_y, beam_width, beam_height)

        # Put the goal near the front-center of the segment
        goal_x = min(L - 1, x1 + max(1, int(0.65 * (x2 - x1))))
        goals[i] = [goal_x, center_y]

        # Carve the gap after the segment unless this is the last one
        if i < 7:
            gap_x1 = x2
            gap_x2 = min(L, x2 + gap_len)
            height_field[gap_x1:gap_x2, :] = pit_depth

            # Add a small landing lip at the beginning of the next segment region
            # to make the transition more realistic and to help with foot placement
            cur_x = gap_x2
        else:
            cur_x = x2

    # Ensure the final region after the last beam is flat pit or ground, not an accidental obstacle
    if cur_x < L:
        height_field[cur_x:, :] = pit_depth

    # Keep the spawn lane flat and continuous
    height_field[:spawn_x, :] = 0.0

    return height_field, goals
```
