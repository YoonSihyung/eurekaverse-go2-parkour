```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A straight-line course of alternating narrow balance beams and low stepped pads over pits."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Terrain grid
    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # Basic dimensions in indices
    L = m_to_idx(length)
    W = m_to_idx(width)
    mid_y = W // 2

    # Spawn-safe flat region
    spawn_x = m_to_idx(2.0)
    height_field[:spawn_x, :] = 0.0

    # Make the rest a pit so the robot must stay on the obstacles
    height_field[spawn_x:, :] = -1.0

    # Course design:
    # Repeated narrow balance-beam / low-pad sequence.
    # The beam forces precise foot placement, while the low pads test stepping
    # and recovery across small changes in support.
    #
    # Difficulty controls beam width, pad size, and spacing slightly.
    beam_width_m = 0.55 - 0.15 * difficulty   # wide enough at easy, narrower at hard
    beam_width = max(m_to_idx(0.40), m_to_idx(beam_width_m))  # never below 0.4 m
    pad_width = max(m_to_idx(1.0), m_to_idx(1.1 - 0.2 * difficulty))
    obstacle_height = 0.02 + 0.10 * difficulty  # small but noticeable elevated support
    beam_length = max(m_to_idx(1.15), m_to_idx(1.35 - 0.2 * difficulty))
    pad_length = max(m_to_idx(0.75), m_to_idx(0.95 - 0.15 * difficulty))
    gap_length = max(m_to_idx(0.35), m_to_idx(0.55 + 0.35 * difficulty))

    # Keep obstacles centered laterally with slight, deterministic offsets to vary foot placement
    lateral_offsets_m = [-0.15, 0.10, -0.05, 0.18, -0.12, 0.08]
    lateral_offsets = [m_to_idx(o) for o in lateral_offsets_m]

    cur_x = spawn_x + m_to_idx(0.4)

    def place_rect(x0, x1, y0, y1, h):
        """Clamp and place a rectangular obstacle patch."""
        x0 = max(0, x0)
        y0 = max(0, y0)
        x1 = min(L, x1)
        y1 = min(W, y1)
        if x1 > x0 and y1 > y0:
            height_field[x0:x1, y0:y1] = h

    # Build 7 traversable sections; place 8 goals along their centers.
    # Goal 0: on the first beam
    for i in range(7):
        # Alternate beam and pad, with a pit gap between each piece
        y_off = lateral_offsets[i % len(lateral_offsets)]
        y_center = int(np.clip(mid_y + y_off, beam_width // 2, W - beam_width // 2 - 1))

        # Beam segment
        x1 = cur_x
        x2 = cur_x + beam_length
        y1 = y_center - beam_width // 2
        y2 = y_center + beam_width // 2
        place_rect(x1, x2, y1, y2, obstacle_height)

        # Goal placed near the middle of the current traversable section
        goals[i] = [x1 + (x2 - x1) // 2, y_center]

        # Advance with a gap over a pit
        cur_x = x2 + gap_length

        # Low stepping pad after the gap, slightly wider than the beam
        pad_y_center = int(np.clip(mid_y - y_off, pad_width // 2, W - pad_width // 2 - 1))
        px1 = cur_x
        px2 = cur_x + pad_length
        py1 = pad_y_center - pad_width // 2
        py2 = pad_y_center + pad_width // 2
        place_rect(px1, px2, py1, py2, obstacle_height * 0.75)

        # Next goal on the pad
        if i < 6:
            goals[i + 1] = [px1 + (px2 - px1) // 2, pad_y_center]

        # Advance past the pad and gap
        cur_x = px2 + gap_length

        # Stop early if we are close to the end
        if cur_x >= L - m_to_idx(1.5):
            break

    # Final goal near the end of the course on a last small beam/pad section
    final_x = min(L - m_to_idx(1.0), max(cur_x, spawn_x + m_to_idx(2.5)))
    final_width = max(m_to_idx(0.8), beam_width)
    final_y = mid_y
    place_rect(final_x, min(L, final_x + m_to_idx(0.9)), final_y - final_width // 2, final_y + final_width // 2, obstacle_height)
    goals[-1] = [final_x + m_to_idx(0.45), final_y]

    # Ensure goals are integer indices and within bounds
    goals[:, 0] = np.clip(np.round(goals[:, 0]), 0, L - 1)
    goals[:, 1] = np.clip(np.round(goals[:, 1]), 0, W - 1)

    return height_field, goals
```
