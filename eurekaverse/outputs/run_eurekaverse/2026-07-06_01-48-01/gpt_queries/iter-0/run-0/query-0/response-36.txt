import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A sequence of narrow elevated beams separated by pits, testing balance and precise stepping."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Terrain grid
    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # ---- Global setup ----
    L = m_to_idx(length)
    W = m_to_idx(width)
    mid_y = W // 2

    # Keep a safe flat spawn zone so the robot does not start on an obstacle
    spawn_len = m_to_idx(2.0)
    height_field[:spawn_len, :] = 0.0

    # Difficulty-scaled beam parameters
    # Wider and slightly lower at easy, narrower and taller at hard
    beam_width_m = 1.25 - 0.45 * difficulty
    beam_width_m = max(0.45, beam_width_m)  # allow narrow obstacle in the rare exception case
    beam_width = max(1, m_to_idx(beam_width_m))

    beam_height = 0.12 + 0.22 * difficulty
    beam_height = float(beam_height)

    # Gap between beams, scaled with difficulty
    gap_m = 0.45 + 0.55 * difficulty
    gap = max(1, m_to_idx(gap_m))

    # Small lateral offset to make stepping less repetitive, but still realistic
    max_offset = m_to_idx(0.55 - 0.15 * difficulty)
    max_offset = int(max(0, max_offset))

    # Beam length: enough to stand on and place a goal on each beam
    beam_len_m = 0.9 - 0.15 * difficulty
    beam_len = max(m_to_idx(0.45), m_to_idx(beam_len_m))

    # Use negative pits between beams so the robot must step/jump onto the beam tops
    pit_depth = -0.9 - 0.4 * difficulty

    def add_beam(x1, x2, y_center, height):
        """Adds a narrow raised beam, clipped to terrain bounds."""
        half_w = beam_width // 2
        y1 = max(0, y_center - half_w)
        y2 = min(W, y_center + half_w + 1)
        x1 = max(0, x1)
        x2 = min(L, x2)
        if x1 < x2 and y1 < y2:
            height_field[x1:x2, y1:y2] = height

    def add_pit(x1, x2):
        """Adds a low pit across the full corridor width."""
        x1 = max(0, x1)
        x2 = min(L, x2)
        if x1 < x2:
            height_field[x1:x2, :] = pit_depth

    # ---- Course layout ----
    # Start with flat ground then alternate pits and beams through the middle corridor.
    cur_x = spawn_len + m_to_idx(0.4)
    y_center = mid_y

    # Goal 0: near the end of the spawn zone
    goals[0] = [spawn_len - m_to_idx(0.4), mid_y]

    # Build 7 beam segments, with the final goal placed on a last beam-like landing
    for i in range(7):
        # Alternate small lateral offsets to require careful foot placement without forcing sharp turns
        offset = random.randint(-max_offset, max_offset) if max_offset > 0 else 0
        y_center = int(np.clip(y_center + offset, beam_width // 2 + 1, W - beam_width // 2 - 2))

        # Add the beam
        x1 = cur_x
        x2 = cur_x + beam_len
        add_beam(x1, x2, y_center, beam_height)

        # Place goal on the beam center
        goals[i] = [x1 + beam_len / 2, y_center]

        # Add pit after the beam except after the last beam
        if i < 6:
            pit_start = x2
            pit_end = x2 + gap
            add_pit(pit_start, pit_end)
            cur_x = pit_end
        else:
            cur_x = x2

    # Final segment: a slightly longer flat landing after the last beam
    landing_len = m_to_idx(1.2)
    landing_x1 = min(L - 1, cur_x)
    landing_x2 = min(L, cur_x + landing_len)
    height_field[landing_x1:landing_x2, :] = 0.0

    # Last goal on the landing to finish the sequence
    goals[7] = [min(L - 1, landing_x1 + landing_len / 2), y_center]

    # Ensure spawn area stays flat and obstacle-free
    height_field[:spawn_len, :] = 0.0

    # Clip goals to valid indices
    goals[:, 0] = np.clip(goals[:, 0], 0, L - 1)
    goals[:, 1] = np.clip(goals[:, 1], 0, W - 1)

    return height_field, goals
