```python
import numpy as np
import random

def set_terrain(length, width, field_resolution, difficulty):
    """A narrow balance-and-gap course with repeated low beams and alternating side offsets."""

    def m_to_idx(m):
        """Converts meters to quantized indices."""
        return np.round(m / field_resolution).astype(np.int16) if not (isinstance(m, list) or isinstance(m, tuple)) else [round(i / field_resolution) for i in m]

    # Terrain grid
    height_field = np.zeros((m_to_idx(length), m_to_idx(width)))
    goals = np.zeros((8, 2))

    # Dimensions in indices
    L = m_to_idx(length)
    W = m_to_idx(width)
    mid_y = W // 2

    # Spawn-safe flat area
    spawn_x = m_to_idx(2.0)
    height_field[:spawn_x, :] = 0.0

    # Main skill: careful foot placement across repeated narrow beams over pits,
    # with alternating lateral offsets to require stable turning and centering.
    # Difficulty increases beam slenderness, gap size, and beam height.
    beam_length_m = 1.15 - 0.25 * difficulty
    beam_length = max(m_to_idx(beam_length_m), m_to_idx(0.8))

    beam_width_m = 1.1 - 0.45 * difficulty
    beam_width = max(m_to_idx(beam_width_m), m_to_idx(0.4))

    gap_length_m = 0.45 + 0.55 * difficulty
    gap_length = max(m_to_idx(gap_length_m), m_to_idx(0.25))

    beam_height = 0.05 + 0.18 * difficulty

    # The course stays mostly as a pit, with beams standing above it.
    pit_height = -0.55 - 0.35 * difficulty
    height_field[spawn_x:, :] = pit_height

    # Keep a safe corridor on the very sides so the terrain still fits well,
    # but the intended path is through the central offset beams.
    side_margin = m_to_idx(0.35)
    height_field[:, :side_margin] = pit_height
    height_field[:, W - side_margin:] = pit_height

    # Beam placement parameters
    cur_x = spawn_x + m_to_idx(0.2)
    y_offsets_m = [0.0, 0.45, -0.35, 0.40, -0.30, 0.32, -0.42, 0.0]
    y_offsets = [m_to_idx(v) for v in y_offsets_m]

    def add_beam(x_start, x_end, y_center):
        """Adds a rectangular beam platform at a given location."""
        y1 = max(0, y_center - beam_width // 2)
        y2 = min(W, y_center + (beam_width + 1) // 2)
        x1 = max(0, x_start)
        x2 = min(L, x_end)
        height_field[x1:x2, y1:y2] = beam_height

    # Create 8 beam segments with small gaps between them.
    for i in range(8):
        y_center = mid_y + y_offsets[i]

        # Clamp beam so it stays within bounds
        y_center = int(np.clip(y_center, beam_width // 2 + 1, W - beam_width // 2 - 2))

        x_start = cur_x
        x_end = min(cur_x + beam_length, L)

        add_beam(x_start, x_end, y_center)

        # Goal placed near the center of each beam segment
        goals[i] = [x_start + (x_end - x_start) / 2.0, y_center]

        # Advance past the beam and a gap
        cur_x = x_end + gap_length

    # If there is remaining terrain, keep it as pit so the last beam is the final objective region.
    if cur_x < L:
        height_field[cur_x:, :] = pit_height

    # Nudge the first goal slightly ahead of spawn, still on flat ground / first transition zone
    goals[0] = [max(spawn_x + m_to_idx(0.35), 0), mid_y]

    # Ensure all goals are valid indices within bounds
    goals[:, 0] = np.clip(goals[:, 0], 0, L - 1)
    goals[:, 1] = np.clip(goals[:, 1], 0, W - 1)

    return height_field, goals
```
